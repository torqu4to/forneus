<script>
  document.querySelectorAll<HTMLElement>('[data-sample]').forEach((sample) => {
    const buttons = [...sample.querySelectorAll<HTMLButtonElement>('[data-sample-choice]')];
    const results = [...sample.querySelectorAll<HTMLElement>('[data-sample-result]')];
    buttons.forEach((button) => button.addEventListener('click', () => {
      const selected = button.dataset.sampleChoice;
      buttons.forEach((choice) => {
        const current = choice.dataset.sampleChoice === selected;
        choice.classList.toggle('is-current', current);
        choice.setAttribute('aria-pressed', String(current));
      });
      results.forEach((result) => { result.hidden = result.dataset.sampleResult !== selected; });
    }));
  });
  if (!window.matchMedia('(prefers-reduced-motion: reduce)').matches && 'IntersectionObserver' in window) {
    const reveals = document.querySelectorAll<HTMLElement>('.cover__top, .cover h1, .cover__words p, .cover .paper-action, .proof h2, .proof__intro>div>p, .proof__aside, .sample__head, .sample__choices, .chapters__heading, .chapter, .way__copy, .way__example, .way__slip, .closing__panel');
    const observer = new IntersectionObserver((entries) => {
      entries.forEach((entry) => {
        if (entry.isIntersecting) {
          entry.target.classList.add('is-visible');
          observer.unobserve(entry.target);
        }
      });
    }, { rootMargin: '0px 0px 90px 0px', threshold: 0.04 });
    reveals.forEach((element) => { element.classList.add('reveal-ready'); observer.observe(element); });
  }
</script>
